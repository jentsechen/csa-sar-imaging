\begin{frame}{Point-Scatterer Echo Model: Geometry}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \vspace{-0.4cm}
      \begin{center}
        \includegraphics[width=\linewidth]{./figures/point_scatterer/point_target_geometry.png}
      \end{center}
    \end{column}
    \begin{column}{0.6\textwidth}
      \begin{itemize}
        \item Target position determines the time-varying slant range $R_k(\eta_m)$.
        \item $R_k(\eta) = \sqrt{R_{0,k}^2 + (\eta_m+\Delta\eta_k)^2 V_r^2}$
        \begin{itemize}
          \item $R_{0,k}\approx R_0+d_k$
          \item $R_0$: slant range to scene center
          \item $d_k$: ground-range offset of scatterer $k$, approximated as slant-range offset
          \item $\eta_m=m\Delta\eta$: slow time
        \item $\Delta\eta_k$: azimuth-time offset of scatterer $k$
        \end{itemize}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}
  \frametitle{Point-Scatterer Echo Model: Signal Synthesis}
  \begin{itemize}
    \item Geometry-Dependent Delay and Phase
    \begin{itemize}
      \item $\tau_k(\eta_m)=\frac{2R_k(\eta_m)}{c}$, $\phi_k(\eta_m)=-4\pi\frac{R_k(\eta_m)}{\lambda}$
    \end{itemize}
    \item Echo Synthesis: $\mathbf{S} = [s_{m,n}]_{M\times N}$
    $$
      s_{m,n} = \sum_k A_k\,\mathrm{rect}\!\left(\frac{\tau_n - \tau_k(\eta_m)}{T_r}\right)
            \cdot e^{j\pi K_r\left(\tau_n - \tau_k(\eta_m)\right)^2}
            \cdot e^{j\phi_k(\eta_m)}
    $$
    \begin{itemize}
      \item $\eta_m = m\Delta\eta$: slow time, $\tau_n = n\Delta\tau$: fast time
      \item $A_k$: scattering coefficient of scatterer $k$, $K_r=\frac{B_r}{T_r}$
    \end{itemize}
  \end{itemize}
\end{frame}
}

\begin{frame}{One-Step Sparse Image Refinement}
  \begin{itemize}
    \item Image-domain regularization [4]
    $$
    \hat{\mathbf{H}}=\arg\min_{\mathbf{H}}
    \underbrace{\frac{1}{2}\left\|\mathbf{H}-\mathbf{H}_{\mathrm{CSA}}\right\|_F^2}_{\text{Preserve CSA image}}
    +\lambda\underbrace{\left\|\mathbf{H}\right\|_1}_{\text{Promote sparsity}}
    $$
    \vspace{-0.4cm}
    \begin{itemize}
      \item Strong scatterers $\rightarrow$ retained; weak components $\rightarrow$ suppressed
      \item $\hat{\mathbf{H}}=\mathcal{S}_{\lambda}(\mathbf{H}_{CSA})$,
            $\mathcal{S}_{\lambda}(x)=\frac{x}{|x|}\max(|x|-\lambda,0),x\neq0$
    \end{itemize}
    \item One-Step Refinement
      \begin{itemize}
        \item $\mathbf{Y}\;\xrightarrow{\mathrm{CSA}}\;\mathbf{H}_{\mathrm{CSA}}
              \;\xrightarrow{\mathcal{S}_{\lambda}(.)}\;\hat{\mathbf{H}}$
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Conclusion}
  \begin{itemize}
    \item Established an end-to-end framework for evaluating SAR image enhancement through ship detection.
    \item One-step sparse refinement suppresses background artifacts while preserving dominant ship responses.

  \end{itemize}
