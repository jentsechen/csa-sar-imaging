% ── d. Problem and Motivation ───────────────────────────────────────────────────────────
\begin{frame}{Problem and Motivation}
  \begin{itemize}
    \item Ground-Based Processing
    \begin{itemize}
      \item Large raw-data volume
      \item Limited downlink bandwidth
      \item Delayed target information
    \end{itemize}
    \item Onboard Processing
    \begin{itemize}
      \item Reduced data transmission and detection latency
      \item Challenge: Limited computation, memory, and power
    \end{itemize}
    \item Research Goal
    \begin{itemize}
      \item Lightweight SAR processing for low-latency ship detection
    \end{itemize}
  \end{itemize}
\end{frame}

% ── e. Proposed Enhanced SAR Imaging and Evaluation Framework ────────────────
% Edit the bullet text below, and uncomment + point \includegraphics at your
% own figure (put image files in the figures/ folder next to this report.tex).
\begin{frame}{Evaluation Framework}
  \begin{itemize}
    \item Research Focus: Enhanced SAR imaging
    \item Evaluation Task: YOLO-based ship detection
  \end{itemize}
  \includegraphics[width=1.05\textwidth]{./figures/evaluation_framework.png}
  \vspace{-0.4cm}
  \begin{itemize}
    \item Evaluation Metrics
    \begin{itemize}
      \item Precision: $\frac{TP}{TP+FP}$
      \item Recall: $\frac{TP}{TP+FN}$
      \item mAP@0.5: Mean AP across classes at IoU threshold = 0.5

      {\footnotesize TP: True Positive, FP: False Positive, FN: False Negative}
    \end{itemize}
  \end{itemize}
\end{frame}
