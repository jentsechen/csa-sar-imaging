    \end{itemize}

\begin{frame}{Point-Scatterer Echo Model: Geometry}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \vspace{-0.4cm}
      \begin{center}
        \includegraphics[width=\linewidth]{./figures/point_target_geometry.png}
      \end{center}
    \end{column}
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Target position determines the time-varying slant range $R_k(\eta_m)$.
        \item $R_k(\eta) = \sqrt{R_{0,k}^2 + (\eta_m+\Delta\eta_k)^2 V_r^2}$
        \begin{itemize}
          \item $R_{0,k}\approx R_0+d_k$
          \item $R_0$: slant range to scene center
          \item $d_k$: range offset of scatterer $k$
          \item $\eta_m=m\Delta\eta$: slow time
        \item $\Delta\eta_k$: azimuth-time offset of scatterer $k$
        \end{itemize}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

  %   \cdot e^{j\pi \frac{B_r}{T_r}\left(\tau - \frac{2R_k(\eta)}{c}\right)^2}

  % \begin{itemize}
  %   \item $\eta = m\Delta\eta$: slow time

  %   \item $A_k$: scattering coefficient of point target $k$

  % \end{itemize}
  
  \begin{itemize}
    \item Geometry-Dependent Delay and Phase
    \begin{itemize}
      \item $\tau_k(\eta_m)=\frac{2R_k(\eta_m)}{c}$
      \item $\phi_k(\eta_m)=-4\pi\frac{R_k(\eta_m)}{\lambda}$
    \end{itemize}
    \item Echo Synthesis: $S = [s_{m,n}]_{M\times N}$
    \begin{itemize}
      \item $s_{m,n} = \sum_k A_k\,\mathrm{rect}\!\left(\frac{\tau_n - \tau_k(\eta_m)}{T_r}\right)
            \cdot e^{j\pi K_r\left(\tau_n - \tau_k(\eta_m)\right)^2}
            \cdot e^{j\phi_k(\eta_m)}$
      \item $\eta_m = m\Delta\eta$: slow time, $\tau_n = n\Delta\tau$: fast time
      \item $A_k$: scattering coefficient of scatterer $k$, $K_r=\frac{B_r}{T_r}$
    \end{itemize}
  \end{itemize}
\end{frame}
